# Python 3.11 (3.10-3.12 also work), tested on Windows 11.
#
#   pip install -r Requirements.txt
#
# GPU notes:
# - The PyTorch wheels below are built for CUDA 12.8 (cu128). RTX 50-series GPUs
#   (compute capability sm_120) need this build; it also runs on older NVIDIA GPUs.
# - MediaPipe has no GPU support on Windows, so keypoint extraction runs on the CPU.

# PyTorch (CUDA 12.8)
--extra-index-url https://download.pytorch.org/whl/cu128
torch==2.11.0+cu128
torchvision==0.26.0+cu128

# Computer vision
# mediapipe is pinned to 0.10.21, the last release with the mediapipe.solutions API
# used by FeatureExtraction/Scripts/keypoint_extractor.py.
# opencv-contrib-python 4.12+ requires numpy>=2, which conflicts with mediapipe 0.10.21.
# Do not also install opencv-python; two OpenCV builds conflict on Windows.
opencv-contrib-python==4.11.0.86
mediapipe==0.10.21
Pillow>=10.0.0

# Numerics (mediapipe 0.10.21 requires numpy<2)
numpy==1.26.4
scipy>=1.13.0,<2.3

# mediapipe 0.10.21 requires protobuf<5; tensorboard 2.21+ requires protobuf>=6.31
protobuf>=4.25.3,<5

# Visualization
matplotlib>=3.8.0
seaborn>=0.13.0

# ML utilities
scikit-learn>=1.4.0
tqdm>=4.65.0

# Notebooks
jupyter>=1.0.0
ipykernel>=6.0.0
ipywidgets>=8.0.0

# ONNX export and validation
onnx>=1.17.0
onnxruntime>=1.20.0
# onnxscript is needed by the dynamo exporter used for MViTv2
onnxscript>=0.7.0

# Experiment tracking (capped because of the protobuf pin above)
tensorboard>=2.16.0,<2.21.0
